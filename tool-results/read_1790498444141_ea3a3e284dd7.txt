    66→/** Shared look for the info cards (location / hours / phone). */
    67→function InfoCard({
    68→  icon,
    69→  tone,
    70→  children,
    71→  as = "div",
    72→  href,
    73→  ariaLabel,
    74→}: {
    75→  icon: React.ReactNode;
    76→  tone: "primary" | "open" | "closed" | "neutral";
    77→  children: React.ReactNode;
    78→  as?: "div" | "a";
    79→  href?: string;
    80→  ariaLabel?: string;
    81→}) {
    82→  const toneClasses = {
    83→    primary: "bg-orange-50 text-primary",
    84→    open: "bg-emerald-50 text-emerald-600",
    85→    closed: "bg-rose-50 text-rose-500",
    86→    neutral: "bg-stone-100 text-stone-400",
    87→  }[tone];
    88→  const className =
    89→    "flex items-start gap-3 rounded-2xl border border-border/70 bg-card p-3.5 shadow-sm";
    90→  if (as === "a" && href) {
    91→    return (
    92→      <a href={href} aria-label={ariaLabel} className={cn(className, "transition-colors hover:bg-accent/70 active:bg-accent")}>
    93→        <span className={cn("grid size-9 shrink-0 place-items-center rounded-xl", toneClasses)}>
    94→          {icon}
    95→        </span>
    96→        {children}
    97→      </a>
    98→    );
    99→  }
   100→  return (
   101→    <div className={className}>
   102→      <span className={cn("grid size-9 shrink-0 place-items-center rounded-xl", toneClasses)}>
   103→        {icon}
   104→      </span>
   105→      {children}
   106→    </div>
   107→  );
   108→}

   110→/**
   111→ * Compact nutrition caption from the dishes table's real columns —
   112→ * only the values that actually exist are included (never "0 kcal").
   113→ */
   114→function nutritionLine(d: Dish): string | null {
   115→  const parts: string[] = [];
   116→  if (d.calories != null) parts.push(`${Math.round(d.calories)} kcal`);
   117→  if (d.protein != null) parts.push(`P ${d.protein} g`);
   118→  if (d.carbs != null) parts.push(`C ${d.carbs} g`);
   119→  if (d.fat != null) parts.push(`F ${d.fat} g`);
   120→  return parts.length > 0 ? parts.join(" · ") : null;
   121→}
