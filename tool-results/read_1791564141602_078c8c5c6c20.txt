    67→/** Shared look for the info cards (location / hours / phone). */
    68→function InfoCard({
    69→  icon,
    70→  tone,
    71→  children,
    72→  as = "div",
    73→  href,
    74→  ariaLabel,
    75→}: {
    76→  icon: React.ReactNode;
    77→  tone: "primary" | "open" | "closed" | "neutral";
    78→  children: React.ReactNode;
    79→  as?: "div" | "a";
    80→  href?: string;
    81→  ariaLabel?: string;
    82→}) {
    83→  const toneClasses = {
    84→    primary: "bg-orange-50 text-primary",
    85→    open: "bg-emerald-50 text-emerald-600",
    86→    closed: "bg-rose-50 text-rose-500",
    87→    neutral: "bg-stone-100 text-stone-400",
    88→  }[tone];
    89→  const className =
    90→    "flex items-start gap-3 rounded-2xl border border-border/70 bg-card p-3.5 shadow-sm";
    91→  if (as === "a" && href) {
    92→    return (
    93→      <a href={href} aria-label={ariaLabel} className={cn(className, "transition-colors hover:bg-accent/70 active:bg-accent")}>
    94→        <span className={cn("grid size-9 shrink-0 place-items-center rounded-xl", toneClasses)}>
    95→          {icon}
    96→        </span>
    97→        {children}
    98→      </a>
    99→    );
   100→  }
   101→  return (
   102→    <div className={className}>
   103→      <span className={cn("grid size-9 shrink-0 place-items-center rounded-xl", toneClasses)}>
   104→        {icon}
   105→      </span>
   106→      {children}
   107→    </div>
   108→  );
   109→}

   111→/**
   112→ * Compact nutrition caption from the dishes table's real columns —
   113→ * only the values that actually exist are included (never "0 kcal").
   114→ */
   115→function nutritionLine(d: Dish): string | null {
   116→  const parts: string[] = [];
   117→  if (d.calories != null) parts.push(`${Math.round(d.calories)} kcal`);
   118→  if (d.protein != null) parts.push(`P ${d.protein} g`);
   119→  if (d.carbs != null) parts.push(`C ${d.carbs} g`);
   120→  if (d.fat != null) parts.push(`F ${d.fat} g`);
   121→  return parts.length > 0 ? parts.join(" · ") : null;
   122→}
