    58→function greeting(): string {
    59→  const h = new Date().getHours();
    60→  if (h < 5) return "Late night cravings";
    61→  if (h < 12) return "Good morning";
    62→  if (h < 17) return "Good afternoon";
    63→  return "Good evening";
    64→}

    66→/**
    67→ * Distance sort that tolerates places WITHOUT a distance (NULL coordinates):
    68→ * they sink to the end of the list instead of faking a number. Only places
    69→ * with a real Haversine distance can ever rank first.
    70→ */
    71→function byKnownDistance(a: PlaceMaybeWithDistance, b: PlaceMaybeWithDistance): number {
    72→  return (
    73→    (a.distanceKm ?? Number.POSITIVE_INFINITY) -
    74→    (b.distanceKm ?? Number.POSITIVE_INFINITY)
    75→  );
    76→}

   188→/** Screen 5 — the discovery home feed, powered by the user's real location */
   189→export function HomeScreen() {
   190→  const go = useFoodieStore((s) => s.go);
   191→  const userName = useFoodieStore((s) => s.userName);
   192→  const eatContext = useFoodieStore((s) => s.eatContext);
   193→  const preferences = useFoodieStore((s) => s.preferences);
   194→  const searchQuery = useFoodieStore((s) => s.searchQuery);
   195→  const setSearchQuery = useFoodieStore((s) => s.setSearchQuery);
   196→  const activeCategory = useFoodieStore((s) => s.activeCategory);
   197→  const setActiveCategory = useFoodieStore((s) => s.setActiveCategory);
   198→  const openPlace = useFoodieStore((s) => s.openPlace);
   199→  const distanceFilterKm = useFoodieStore((s) => s.distanceFilterKm);
   200→  const setDistanceFilter = useFoodieStore((s) => s.setDistanceFilter);
   201→  const sortBy = useFoodieStore((s) => s.sortBy);
   202→  const setSortBy = useFoodieStore((s) => s.setSortBy);
   203→
   204→  // --- real location state (native Geolocation API → Zustand) ---
   205→  const headline = useLocationHeadline();
   206→  const locSource = useLocationStore((s) => s.source);
   207→  const locStatus = useLocationStore((s) => s.status);
   208→  const origin = useLocationStore((s) => s.origin);
   209→
   210→  const [expanded, setExpanded] = useState<string | null>(null);
   211→  const [locSheetOpen, setLocSheetOpen] = useState(false);
   212→
   213→  const searching = searchQuery.trim().length > 0 || activeCategory !== null;
   214→
   215→  /* 0) Restaurant data — two read-only sources:
   216→   *    - nearby feed: coordinate-bearing places, distance-annotated server-side;
   217→   *    - full list:   every discoverable place, INCLUDING rows whose
   218→   *      coordinates are still NULL ("Location pending verification").
   219→   *    Neither source ever writes to the database. */
   220→  const feed = useRestaurantFeed(origin);
   221→  const allFeed = useAllRestaurants();
   222→
   223→  /* 1) Haversine distance from the active origin to every coordinate-bearing
   224→   *    place. NULL-coordinate rows never enter this math — they are kept as-is
   225→   *    (NO distanceKm) and tracked separately as "pending". */
   226→  const withDistance = useMemo(() => withDistances(feed.places, origin), [feed.places, origin]);
   227→  const allAnnotated = useMemo(
   228→    () =>
   229→      allFeed.places.map((p) =>
   230→        p.latitude != null && p.longitude != null
   231→          ? { ...p, distanceKm: haversineKm(origin, { lat: p.latitude, lng: p.longitude }) }
   232→          : p
   233→      ),
   234→    [allFeed.places, origin]
   235→  );
   236→
   237→  /* Merge by id — nearby results win (authoritative distance source). */
   238→  const merged = useMemo(() => {
   239→    const byId = new Map<string, PlaceMaybeWithDistance>();
   240→    for (const p of allAnnotated) byId.set(p.id, p);
   241→    for (const p of withDistance) byId.set(p.id, p);
   242→    return [...byId.values()];
   243→  }, [allAnnotated, withDistance]);
   244→
   245→  /* Places with a KNOWN distance — the only ones eligible for radius
   246→   * filters and "nearest" sorting. */
   247→  const withKnownDistance = useMemo(
   248→    () => merged.filter((p): p is PlaceWithDistance => p.distanceKm != null),
   249→    [merged]
   250→  );
   251→
   252→  /* NULL-coordinate places — honest "Location pending verification" rows.
   253→   * They carry no distance and never fake one. */
   254→  const pending = useMemo(() => merged.filter((p) => p.distanceKm == null), [merged]);
   255→
   256→  /* 2) Distance filter applies ONLY to places with known distances. */
   257→  const inRange = useMemo(
   258→    () =>
   259→      distanceFilterKm == null
   260→        ? withKnownDistance
   261→        : withKnownDistance.filter((p) => p.distanceKm <= distanceFilterKm),
   262→    [withKnownDistance, distanceFilterKm]
   263→  );
   264→
   265→  /* 3) Sort + section the filtered list. */
   266→  const nearby = useMemo(
   267→    () => (sortBy === "nearest" ? [...inRange].sort(byDistance) : [...inRange].sort(byRating)),
   268→    [inRange, sortBy]
   269→  );
   270→  const gems = useMemo(() => hiddenGems(inRange), [inRange]);
   271→  const favorites = useMemo(() => localFavorites(inRange), [inRange]);
   272→  const popular = useMemo(() => popularPlaces(inRange), [inRange]);
   273→  const budget = useMemo(() => budgetEats(inRange), [inRange]);
   274→
   275→  /* 4) Search / category results — pending places join the pool whenever no
   276→   *    radius filter is active (an unknown distance can't honestly satisfy
   277→   *    "within X km"). With "Nearest First" they sort to the very end. */
   278→  const results = useMemo(() => {
   279→    const pool: PlaceMaybeWithDistance[] =
   280→      distanceFilterKm == null ? [...inRange, ...pending] : [...inRange];
   281→    let list = pool;
   282→    if (activeCategory) {
   283→      list = list.filter((p) => p.categoryIds.includes(activeCategory));
   284→    }
   285→    if (searchQuery.trim()) {
   286→      list = list.filter((p) => matchesQuery(p, searchQuery));
   287→    }
   288→    return sortBy === "nearest" ? [...list].sort(byKnownDistance) : [...list].sort(byRating);
   289→  }, [inRange, pending, distanceFilterKm, activeCategory, searchQuery, sortBy]);
   290→
   291→  /* Page-level loading / error across BOTH feeds (never a silent partial). */
   292→  const feedLoading = feed.status === "loading" || allFeed.status === "loading";
   293→  const feedError =
   294→    feed.status === "error" ? feed.error : allFeed.status === "error" ? allFeed.error : null;
   295→  const retryFeed = () => {
   296→    feed.reload();
   297→    allFeed.reload();
   298→  };
   299→
   300→  const discoveryEmpty =
   301→    nearby.length === 0 &&
   302→    gems.length === 0 &&
   303→    favorites.length === 0 &&
   304→    popular.length === 0 &&
   305→    budget.length === 0 &&
   306→    // pending places count as content only when no radius filter hides them
   307→    (distanceFilterKm != null || pending.length === 0);
   308→
   309→  /* The backend answered, but the database truly has zero restaurants. */
   310→  const dbEmpty =
   311→    feed.status === "ready" && allFeed.status === "ready" && merged.length === 0;
   312→
   313→  const demoBanner = feed.mode === "demo" || allFeed.mode === "demo";
   314→  const demoNotice = feed.notice ?? allFeed.notice;
   315→
   316→  const preferredCats = PLACE_CATEGORIES.filter((c) => preferences.includes(c.id));
   317→
   318→  const nearbySubtitle =
   319→    locSource === "browser"
   320→      ? "Closest to your live location"
   321→      : locSource === "manual"
   322→        ? "Distances from your selected city"
   323→        : locSource === "test"
   324→          ? `Distances from ${GITAM_TEST_LOCATION.label}`
   325→          : `Around ${PROTOTYPE_CITY.area} · prototype origin`;
   326→
   327→  function toggleExpand(key: string) {
   328→    setExpanded((cur) => (cur === key ? null : key));
   329→  }
   330→
   331→  return (
   332→    <div className="relative flex h-full flex-col bg-background">
   333→      {/* ---------------- header ---------------- */}
   334→      <header className="z-10 shrink-0 border-b border-border/50 bg-background/95 px-5 pb-3 pt-5 backdrop-blur supports-[backdrop-filter]:bg-background/80">
   335→        <div className="flex items-center justify-between gap-3">
   336→          <button
   337→            type="button"
   338→            onClick={() => setLocSheetOpen(true)}
   339→            aria-haspopup="dialog"
   340→            className="group flex min-w-0 items-start gap-1.5 text-left"
   341→          >
   342→            <span className="relative mt-0.5 shrink-0">
   343→              <MapPinned
   344→                className={cn(
   345→                  "size-4",
   346→                  headline.tone === "live" ? "text-emerald-600" : "text-primary"
   347→                )}
   348→                aria-hidden="true"
   349→              />
   350→              {headline.tone === "live" && (
   351→                <span
   352→                  className="absolute -right-0.5 -top-0.5 size-1.5 animate-pulse rounded-full bg-emerald-500"
   353→                  aria-hidden="true"
   354→                />
   355→              )}
   356→            </span>
   357→            <span className="min-w-0">
   358→              <span
   359→                className={cn(
   360→                  "block text-[10.5px] font-bold uppercase tracking-wider",
   361→                  headline.tone === "live"
