    60→function greeting(): string {
    61→  const h = new Date().getHours();
    62→  if (h < 5) return "Late night cravings";
    63→  if (h < 12) return "Good morning";
    64→  if (h < 17) return "Good afternoon";
    65→  return "Good evening";
    66→}

    68→/**
    69→ * Distance sort that tolerates places WITHOUT a distance (NULL coordinates):
    70→ * they sink to the end of the list instead of faking a number. Only places
    71→ * with a real Haversine distance can ever rank first.
    72→ */
    73→function byKnownDistance(a: PlaceMaybeWithDistance, b: PlaceMaybeWithDistance): number {
    74→  return (
    75→    (a.distanceKm ?? Number.POSITIVE_INFINITY) -
    76→    (b.distanceKm ?? Number.POSITIVE_INFINITY)
    77→  );
    78→}

   190→/** Screen 5 — the discovery home feed, powered by the user's real location */
   191→export function HomeScreen() {
   192→  const go = useFoodieStore((s) => s.go);
   193→  const userName = useFoodieStore((s) => s.userName);
   194→  const eatContext = useFoodieStore((s) => s.eatContext);
   195→  const preferences = useFoodieStore((s) => s.preferences);
   196→  const searchQuery = useFoodieStore((s) => s.searchQuery);
   197→  const setSearchQuery = useFoodieStore((s) => s.setSearchQuery);
   198→  const activeCategory = useFoodieStore((s) => s.activeCategory);
   199→  const setActiveCategory = useFoodieStore((s) => s.setActiveCategory);
   200→  const openPlace = useFoodieStore((s) => s.openPlace);
   201→  const distanceFilterKm = useFoodieStore((s) => s.distanceFilterKm);
   202→  const setDistanceFilter = useFoodieStore((s) => s.setDistanceFilter);
   203→  const sortBy = useFoodieStore((s) => s.sortBy);
   204→  const setSortBy = useFoodieStore((s) => s.setSortBy);
   205→
   206→  // --- real location state (native Geolocation API → Zustand) ---
   207→  const headline = useLocationHeadline();
   208→  const locSource = useLocationStore((s) => s.source);
   209→  const locStatus = useLocationStore((s) => s.status);
   210→  const origin = useLocationStore((s) => s.origin);
   211→
   212→  const [expanded, setExpanded] = useState<string | null>(null);
   213→  const [locSheetOpen, setLocSheetOpen] = useState(false);
   214→
   215→  const searching = searchQuery.trim().length > 0 || activeCategory !== null;
   216→
   217→  /* 0) Restaurant data — two read-only sources:
   218→   *    - nearby feed: coordinate-bearing places, distance-annotated server-side;
   219→   *    - full list:   every discoverable place, INCLUDING rows whose
   220→   *      coordinates are still NULL ("Location pending verification").
   221→   *    Neither source ever writes to the database. */
   222→  const feed = useRestaurantFeed(origin);
   223→  const allFeed = useAllRestaurants();
   224→
   225→  /* 1) Haversine distance from the active origin to every coordinate-bearing
   226→   *    place. NULL-coordinate rows never enter this math — they are kept as-is
   227→   *    (NO distanceKm) and tracked separately as "pending". */
   228→  const withDistance = useMemo(() => withDistances(feed.places, origin), [feed.places, origin]);
   229→  const allAnnotated = useMemo(
   230→    () =>
   231→      allFeed.places.map((p) =>
   232→        p.latitude != null && p.longitude != null
   233→          ? { ...p, distanceKm: haversineKm(origin, { lat: p.latitude, lng: p.longitude }) }
   234→          : p
   235→      ),
   236→    [allFeed.places, origin]
   237→  );
   238→
   239→  /* Merge by id — nearby results win (authoritative distance source). */
   240→  const merged = useMemo(() => {
   241→    const byId = new Map<string, PlaceMaybeWithDistance>();
   242→    for (const p of allAnnotated) byId.set(p.id, p);
   243→    for (const p of withDistance) byId.set(p.id, p);
   244→    return [...byId.values()];
   245→  }, [allAnnotated, withDistance]);
   246→
   247→  /* Places with a KNOWN distance — the only ones eligible for radius
   248→   * filters and "nearest" sorting. */
   249→  const withKnownDistance = useMemo(
   250→    () => merged.filter((p): p is PlaceWithDistance => p.distanceKm != null),
   251→    [merged]
   252→  );
   253→
   254→  /* NULL-coordinate places — honest "Location pending verification" rows.
   255→   * They carry no distance and never fake one. */
   256→  const pending = useMemo(() => merged.filter((p) => p.distanceKm == null), [merged]);
   257→
   258→  /* 2) Distance filter applies ONLY to places with known distances. */
   259→  const inRange = useMemo(
   260→    () =>
   261→      distanceFilterKm == null
   262→        ? withKnownDistance
   263→        : withKnownDistance.filter((p) => p.distanceKm <= distanceFilterKm),
   264→    [withKnownDistance, distanceFilterKm]
   265→  );
   266→
   267→  /* 3) Sort + section the filtered list. */
   268→  const nearby = useMemo(
   269→    () => (sortBy === "nearest" ? [...inRange].sort(byDistance) : [...inRange].sort(byRating)),
   270→    [inRange, sortBy]
   271→  );
   272→  const gems = useMemo(() => hiddenGems(inRange), [inRange]);
   273→  const favorites = useMemo(() => localFavorites(inRange), [inRange]);
   274→  const popular = useMemo(() => popularPlaces(inRange), [inRange]);
   275→  const budget = useMemo(() => budgetEats(inRange), [inRange]);
   276→
   277→  /* 4) Search / category results — pending places join the pool whenever no
   278→   *    radius filter is active (an unknown distance can't honestly satisfy
   279→   *    "within X km"). With "Nearest First" they sort to the very end. */
   280→  const results = useMemo(() => {
   281→    const pool: PlaceMaybeWithDistance[] =
   282→      distanceFilterKm == null ? [...inRange, ...pending] : [...inRange];
   283→    let list = pool;
   284→    if (activeCategory) {
   285→      list = list.filter((p) => p.categoryIds.includes(activeCategory));
   286→    }
   287→    if (searchQuery.trim()) {
   288→      list = list.filter((p) => matchesQuery(p, searchQuery));
   289→    }
   290→    return sortBy === "nearest" ? [...list].sort(byKnownDistance) : [...list].sort(byRating);
   291→  }, [inRange, pending, distanceFilterKm, activeCategory, searchQuery, sortBy]);
   292→
   293→  /* Page-level loading / error across BOTH feeds (never a silent partial). */
   294→  const feedLoading = feed.status === "loading" || allFeed.status === "loading";
   295→  const feedError =
   296→    feed.status === "error" ? feed.error : allFeed.status === "error" ? allFeed.error : null;
   297→  const retryFeed = () => {
   298→    feed.reload();
   299→    allFeed.reload();
   300→  };
   301→
   302→  const discoveryEmpty =
   303→    nearby.length === 0 &&
   304→    gems.length === 0 &&
   305→    favorites.length === 0 &&
   306→    popular.length === 0 &&
   307→    budget.length === 0 &&
   308→    // pending places count as content only when no radius filter hides them
   309→    (distanceFilterKm != null || pending.length === 0);
   310→
   311→  /* The backend answered, but the database truly has zero restaurants. */
   312→  const dbEmpty =
   313→    feed.status === "ready" && allFeed.status === "ready" && merged.length === 0;
   314→
   315→  const demoBanner = feed.mode === "demo" || allFeed.mode === "demo";
   316→  const demoNotice = feed.notice ?? allFeed.notice;
   317→
   318→  const preferredCats = PLACE_CATEGORIES.filter((c) => preferences.includes(c.id));
   319→
   320→  const nearbySubtitle =
   321→    locSource === "browser"
   322→      ? "Closest to your live location"
   323→      : locSource === "manual"
   324→        ? "Distances from your selected city"
   325→        : locSource === "test"
   326→          ? `Distances from ${GITAM_TEST_LOCATION.label}`
   327→          : `Around ${PROTOTYPE_CITY.area} · prototype origin`;
   328→
   329→  function toggleExpand(key: string) {
   330→    setExpanded((cur) => (cur === key ? null : key));
   331→  }
   332→
   333→  return (
   334→    <div className="relative flex h-full flex-col bg-background">
   335→      {/* ---------------- header ---------------- */}
   336→      <header className="z-10 shrink-0 border-b border-border/50 bg-background/95 px-5 pb-3 pt-5 backdrop-blur supports-[backdrop-filter]:bg-background/80">
   337→        <div className="flex items-center justify-between gap-3">
   338→          <button
   339→            type="button"
   340→            onClick={() => setLocSheetOpen(true)}
   341→            aria-haspopup="dialog"
   342→            className="group flex min-w-0 items-start gap-1.5 text-left"
   343→          >
   344→            <span className="relative mt-0.5 shrink-0">
   345→              <MapPinned
   346→                className={cn(
   347→                  "size-4",
   348→                  headline.tone === "live" ? "text-emerald-600" : "text-primary"
   349→                )}
   350→                aria-hidden="true"
   351→              />
   352→              {headline.tone === "live" && (
   353→                <span
   354→                  className="absolute -right-0.5 -top-0.5 size-1.5 animate-pulse rounded-full bg-emerald-500"
   355→                  aria-hidden="true"
   356→                />
   357→              )}
   358→            </span>
   359→            <span className="min-w-0">
   360→              <span
   361→                className={cn(
   362→                  "block text-[10.5px] font-bold uppercase tracking-wider",
   363→                  headline.tone === "live"
   364→                    ? "text-emerald-600"
