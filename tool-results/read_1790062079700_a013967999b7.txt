    45→function greeting(): string {
    46→  const h = new Date().getHours();
    47→  if (h < 5) return "Late night cravings";
    48→  if (h < 12) return "Good morning";
    49→  if (h < 17) return "Good afternoon";
    50→  return "Good evening";
    51→}

   163→/** Screen 5 — the discovery home feed, powered by the user's real location */
   164→export function HomeScreen() {
   165→  const go = useFoodieStore((s) => s.go);
   166→  const userName = useFoodieStore((s) => s.userName);
   167→  const eatContext = useFoodieStore((s) => s.eatContext);
   168→  const preferences = useFoodieStore((s) => s.preferences);
   169→  const searchQuery = useFoodieStore((s) => s.searchQuery);
   170→  const setSearchQuery = useFoodieStore((s) => s.setSearchQuery);
   171→  const activeCategory = useFoodieStore((s) => s.activeCategory);
   172→  const setActiveCategory = useFoodieStore((s) => s.setActiveCategory);
   173→  const openPlace = useFoodieStore((s) => s.openPlace);
   174→  const distanceFilterKm = useFoodieStore((s) => s.distanceFilterKm);
   175→  const setDistanceFilter = useFoodieStore((s) => s.setDistanceFilter);
   176→  const sortBy = useFoodieStore((s) => s.sortBy);
   177→  const setSortBy = useFoodieStore((s) => s.setSortBy);
   178→
   179→  // --- real location state (native Geolocation API → Zustand) ---
   180→  const headline = useLocationHeadline();
   181→  const locSource = useLocationStore((s) => s.source);
   182→  const locStatus = useLocationStore((s) => s.status);
   183→  const origin = useLocationStore((s) => s.origin);
   184→
   185→  const [expanded, setExpanded] = useState<string | null>(null);
   186→  const [locSheetOpen, setLocSheetOpen] = useState(false);
   187→
   188→  const searching = searchQuery.trim().length > 0 || activeCategory !== null;
   189→
   190→  /* 0) Restaurant data — real Supabase database when connected,
   191→   *    clearly-labelled demo dataset otherwise. Refetches on origin change. */
   192→  const feed = useRestaurantFeed(origin);
   193→
   194→  /* 1) Haversine distance from the active origin to EVERY restaurant. */
   195→  const withDistance = useMemo(() => withDistances(feed.places, origin), [feed.places, origin]);
   196→
   197→  /* 2) Apply the distance filter (within 1/3/5/10 km). */
   198→  const inRange = useMemo(
   199→    () =>
   200→      distanceFilterKm == null
   201→        ? withDistance
   202→        : withDistance.filter((p) => p.distanceKm <= distanceFilterKm),
   203→    [withDistance, distanceFilterKm]
   204→  );
   205→
   206→  /* 3) Sort + section the filtered list. */
   207→  const nearby = useMemo(
   208→    () => (sortBy === "nearest" ? [...inRange].sort(byDistance) : [...inRange].sort(byRating)),
   209→    [inRange, sortBy]
   210→  );
   211→  const gems = useMemo(() => hiddenGems(inRange), [inRange]);
   212→  const favorites = useMemo(() => localFavorites(inRange), [inRange]);
   213→  const popular = useMemo(() => popularPlaces(inRange), [inRange]);
   214→  const budget = useMemo(() => budgetEats(inRange), [inRange]);
   215→
   216→  const results = useMemo(() => {
   217→    let list = inRange;
   218→    if (activeCategory) {
   219→      list = list.filter((p) => p.categoryIds.includes(activeCategory));
   220→    }
   221→    if (searchQuery.trim()) {
   222→      list = list.filter((p) => matchesQuery(p, searchQuery));
   223→    }
   224→    return sortBy === "nearest" ? [...list].sort(byDistance) : [...list].sort(byRating);
   225→  }, [inRange, activeCategory, searchQuery, sortBy]);
   226→
   227→  const discoveryEmpty =
   228→    nearby.length === 0 &&
   229→    gems.length === 0 &&
   230→    favorites.length === 0 &&
   231→    popular.length === 0 &&
   232→    budget.length === 0;
   233→
   234→  /* The backend answered, but there are simply no restaurants to show. */
   235→  const dbEmpty = feed.status === "ready" && feed.places.length === 0;
   236→
   237→  const preferredCats = PLACE_CATEGORIES.filter((c) => preferences.includes(c.id));
   238→
   239→  const nearbySubtitle =
   240→    locSource === "browser"
   241→      ? "Closest to your live location"
   242→      : locSource === "manual"
   243→        ? "Distances from your selected city"
   244→        : `Around ${PROTOTYPE_CITY.area} · prototype origin`;
   245→
   246→  function toggleExpand(key: string) {
   247→    setExpanded((cur) => (cur === key ? null : key));
   248→  }
   249→
   250→  return (
   251→    <div className="relative flex h-full flex-col bg-background">
   252→      {/* ---------------- header ---------------- */}
   253→      <header className="z-10 shrink-0 border-b border-border/50 bg-background/95 px-5 pb-3 pt-5 backdrop-blur supports-[backdrop-filter]:bg-background/80">
   254→        <div className="flex items-center justify-between gap-3">
   255→          <button
   256→            type="button"
   257→            onClick={() => setLocSheetOpen(true)}
   258→            aria-haspopup="dialog"
   259→            className="group flex min-w-0 items-start gap-1.5 text-left"
   260→          >
   261→            <span className="relative mt-0.5 shrink-0">
   262→              <MapPinned
   263→                className={cn(
   264→                  "size-4",
   265→                  headline.tone === "live" ? "text-emerald-600" : "text-primary"
   266→                )}
   267→                aria-hidden="true"
   268→              />
   269→              {headline.tone === "live" && (
   270→                <span
   271→                  className="absolute -right-0.5 -top-0.5 size-1.5 animate-pulse rounded-full bg-emerald-500"
   272→                  aria-hidden="true"
   273→                />
   274→              )}
   275→            </span>
   276→            <span className="min-w-0">
   277→              <span
   278→                className={cn(
   279→                  "block text-[10.5px] font-bold uppercase tracking-wider",
   280→                  headline.tone === "live"
   281→                    ? "text-emerald-600"
   282→                    : headline.tone === "warn"
   283→                      ? "text-amber-600"
   284→                      : "text-muted-foreground"
   285→                )}
   286→              >
   287→                {headline.kicker}
   288→              </span>
   289→              <span className="flex items-center gap-0.5 text-[14px] font-bold leading-tight text-foreground">
   290→                <span className="truncate">{headline.value}</span>
   291→                <ChevronDown
   292→                  className="size-3.5 shrink-0 text-muted-foreground transition-transform group-hover:translate-y-0.5"
   293→                  aria-hidden="true"
   294→                />
   295→              </span>
   296→            </span>
   297→          </button>
   298→
   299→          <button
   300→            type="button"
   301→            onClick={() =>
   302→              toast.info("No new alerts", { description: "Foodie tips will show up here soon." })
   303→            }
   304→            aria-label="Notifications"
   305→            className="relative grid size-9 shrink-0 place-items-center rounded-full border border-border bg-card shadow-sm transition-colors hover:bg-accent"
   306→          >
   307→            <Bell className="size-4" />
   308→            <span className="absolute right-2 top-2 size-1.5 rounded-full bg-primary" aria-hidden="true" />
   309→          </button>
   310→        </div>
   311→
   312→        <SearchBar value={searchQuery} onChange={setSearchQuery} className="mt-3.5" />
   313→
   314→        {/* food categories */}
   315→        <div className="no-scrollbar -mx-5 mt-3 flex gap-2 overflow-x-auto px-5" role="tablist" aria-label="Food categories">
   316→          <CategoryChip
   317→            category={{ id: "all", label: "All", emoji: "🍽️" }}
   318→            active={activeCategory === null}
   319→            onClick={() => {
   320→              setActiveCategory(null);
   321→              setSearchQuery("");
   322→            }}
   323→          />
   324→          {PLACE_CATEGORIES.map((c) => (
   325→            <CategoryChip
   326→              key={c.id}
   327→              category={c}
   328→              active={activeCategory === c.id}
   329→              onClick={() => setActiveCategory(activeCategory === c.id ? null : c.id)}
   330→            />
   331→          ))}
   332→        </div>
   333→
   334→        {/* distance filter + sorting */}
   335→        <DiscoveryFilterBar
   336→          distanceFilterKm={distanceFilterKm}
   337→          onDistanceChange={setDistanceFilter}
   338→          sortBy={sortBy}
   339→          onSortChange={setSortBy}
   340→          className="mt-3"
   341→        />
   342→      </header>
   343→
   344→      {/* ---------------- feed ---------------- */}
   345→      <main className="thin-scrollbar flex-1 overflow-y-auto pb-10" role="main">
   346→        {feed.status === "loading" && <FeedSkeleton />}
   347→
   348→        {feed.status === "error" && (
   349→          <FeedErrorCard error={feed.error} onRetry={feed.reload} />
   350→        )}
   351→
   352→        {feed.status === "ready" && (
   353→          <>
   354→            {feed.mode === "demo" && (
   355→              <div className="px-5 pt-4">
   356→                <DemoDataBanner notice={feed.notice} />
   357→              </div>
   358→            )}
   359→        {searching ? (
   360→          /* ------- search / filter results ------- */
   361→          <section className="px-5 pt-5">
   362→            <h2 className="text-[16px] font-bold text-foreground">
   363→              {results.length} {results.length === 1 ? "place" : "places"}{" "}
   364→              {searchQuery.trim() ? (
   365→                <>
   366→                  for <span className="text-primary">&ldquo;{searchQuery.trim()}&rdquo;</span>
   367→                </>
   368→              ) : (
   369→                <>
   370→                  in{" "}
   371→                  <span className="text-primary">
   372→                    {PLACE_CATEGORIES.find((c) => c.id === activeCategory)?.label}
   373→                  </span>
   374→                </>
   375→              )}
   376→              {distanceFilterKm != null && (
   377→                <span className="text-muted-foreground"> · within {distanceFilterKm} km</span>
   378→              )}
   379→            </h2>
   380→            {preferredCats.length > 0 && !searchQuery.trim() && (
   381→              <p className="mt-0.5 text-[12.5px] text-muted-foreground">
   382→                Based on your love for {preferredCats.map((c) => c.label).join(", ")}
   383→              </p>
   384→            )}
   385→            <div className="mt-4 space-y-2.5">
   386→              {results.map((p) => (
   387→                <ListRow key={p.id} place={p} onSelect={openPlace} />
   388→              ))}
   389→              {results.length === 0 && (
   390→                <div className="flex flex-col items-center rounded-2xl border border-dashed border-border bg-card/60 px-6 py-10 text-center">
   391→                  <SearchX className="size-9 text-stone-300" aria-hidden="true" />
   392→                  <p className="mt-3 text-[15px] font-bold text-foreground">Nothing matches… yet</p>
   393→                  <p className="mt-1 text-[13px] text-muted-foreground">
   394→                    {distanceFilterKm != null
   395→                      ? `Nothing within ${distanceFilterKm} km — try widening the distance.`
   396→                      : "Try \u201Cbiryani\u201D, \u201Cdosa\u201D, \u201Ccoffee\u201D or clear the filters."}
   397→                  </p>
   398→                  <div className="mt-4 flex gap-2">
   399→                    {distanceFilterKm != null && (
   400→                      <button
   401→                        type="button"
   402→                        onClick={() => setDistanceFilter(null)}
   403→                        className="rounded-full border border-border bg-card px-4 py-2 text-[13px] font-bold text-foreground transition-colors hover:bg-accent"
   404→                      >
   405→                        Any distance
   406→                      </button>
   407→                    )}
   408→                    <button
   409→                      type="button"
   410→                      onClick={() => {
   411→                        setSearchQuery("");
   412→                        setActiveCategory(null);
   413→                      }}
   414→                      className="rounded-full bg-primary px-4 py-2 text-[13px] font-bold text-white shadow-md shadow-orange-600/25 transition-all hover:bg-primary/90 active:scale-95"
   415→                    >
   416→                      Reset search
   417→                    </button>
   418→                  </div>
   419→                </div>
   420→              )}
   421→            </div>
   422→          </section>
   423→        ) : (
   424→          /* ------- discovery sections ------- */
   425→          <>
   426→            {/* 📍 Use my current location — clear GPS control */}
   427→            <div className="mx-5 mt-5">
   428→              <LocationPromptBanner onOpen={() => setLocSheetOpen(true)} />
   429→            </div>
   430→
   431→            {/* greeting */}
   432→            <motion.section
   433→              initial={{ opacity: 0, y: 10 }}
   434→              animate={{ opacity: 1, y: 0 }}
   435→              className="mx-5 mt-4 overflow-hidden rounded-3xl foodie-gradient-soft p-4"
   436→            >
   437→              <div className="flex items-center justify-between gap-3">
   438→                <div className="min-w-0">
   439→                  <p className="text-[15px] font-extrabold tracking-tight text-foreground">
   440→                    {greeting()}
   441→                    {userName ? `, ${userName}` : ""}! <span aria-hidden="true">🌆</span>
   442→                  </p>
   443→                  <p className="mt-0.5 truncate text-[12.5px] text-muted-foreground">
   444→                    {eatContext
   445→                      ? `Showing picks great for ${eatContext.toLowerCase()} dining`
   446→                      : "Hungry? Let's find your next favorite spot."}
   447→                  </p>
   448→                </div>
   449→                <span className="text-3xl" aria-hidden="true">
   450→                  {eatContext === "Travelling" ? "🧳" : eatContext === "Family" ? "👨‍👩‍👧" : eatContext === "Partner" ? "❤️" : eatContext === "Friends" ? "🎉" : "🍽️"}
   451→                </span>
   452→              </div>
   453→              {preferences.length > 0 && (
   454→                <div className="no-scrollbar mt-3 flex gap-1.5 overflow-x-auto" aria-label="Your preferences">
   455→                  {preferredCats.map((c) => (
   456→                    <span
   457→                      key={c.id}
   458→                      className="shrink-0 rounded-full bg-white/80 px-2.5 py-1 text-[11px] font-semibold text-stone-700 shadow-sm"
   459→                    >
   460→                      {c.emoji} {c.label}
   461→                    </span>
   462→                  ))}
   463→                </div>
   464→              )}
   465→            </motion.section>
   466→
   467→            {dbEmpty ? (
   468→              <EmptyDatabaseCard mode={feed.mode} />
   469→            ) : discoveryEmpty ? (
   470→              /* everything filtered out by the distance radius */
   471→              <div className="mx-5 mt-6 flex flex-col items-center rounded-3xl border border-dashed border-border bg-card/60 px-6 py-12 text-center">
   472→                <MapPinned className="size-10 text-stone-300" aria-hidden="true" />
   473→                <p className="mt-3 text-[15px] font-bold text-foreground">
   474→                  {distanceFilterKm != null
   475→                    ? `No places within ${distanceFilterKm} km`
   476→                    : "No places to show right now"}
   477→                </p>
   478→                <p className="mt-1 max-w-[240px] text-[13px] leading-snug text-muted-foreground">
   479→                  Widen the radius to see everything Foodie has around this origin.
   480→                </p>
   481→                <button
   482→                  type="button"
   483→                  onClick={() => setDistanceFilter(null)}
   484→                  className="mt-4 rounded-full bg-primary px-4 py-2 text-[13px] font-bold text-white shadow-md shadow-orange-600/25 transition-all hover:bg-primary/90 active:scale-95"
   485→                >
   486→                  Show any distance
   487→                </button>
   488→              </div>
   489→            ) : (
   490→              <>
   491→                {/* nearby */}
   492→                {nearby.length > 0 && (
   493→                  <section className="mt-7">
   494→                    <SectionHeader
   495→                      emoji="📍"
   496→                      title="Nearby food places"
   497→                      subtitle={nearbySubtitle}
   498→                      className="px-5"
   499→                    />
   500→                    <div className="space-y-2.5 px-5">
   501→                      {nearby.slice(0, 4).map((p) => (
   502→                        <ListRow key={p.id} place={p} onSelect={openPlace} />
   503→                      ))}
   504→                    </div>
   505→                  </section>
   506→                )}
   507→
   508→                {/* hidden gems */}
   509→                {gems.length > 0 && (
   510→                  <RailSection
   511→                    emoji="💎"
   512→                    title="Hidden Gems"
   513→                    subtitle="Low-key spots locals swear by"
   514→                    total={gems.length}
   515→                    expanded={expanded === "gems"}
   516→                    onSeeAll={() => toggleExpand("gems")}
   517→                  >
   518→                    {gems.map((p) =>
   519→                      expanded === "gems" ? (
   520→                        <ListRow key={p.id} place={p} onSelect={openPlace} />
   521→                      ) : (
   522→                        <CarouselCard key={p.id} place={p} onSelect={openPlace} />
   523→                      )
   524→                    )}
