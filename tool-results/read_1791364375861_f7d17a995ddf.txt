    60→    60→function greeting(): string {
    61→    61→  const h = new Date().getHours();
    62→    62→  if (h < 5) return "Late night cravings";
    63→    63→  if (h < 12) return "Good morning";
    64→    64→  if (h < 17) return "Good afternoon";
    65→    65→  return "Good evening";

    68→    68→/**
    69→    69→ * Distance sort that tolerates places WITHOUT a distance (NULL coordinates):
    70→    70→ * they sink to the end of the list instead of faking a number. Only places
    71→    71→ * with a real Haversine distance can ever rank first.
    72→    72→ */
    73→    73→function byKnownDistance(a: PlaceMaybeWithDistance, b: PlaceMaybeWithDistance): number {
    74→    74→  return (
    75→    75→    (a.distanceKm ?? Number.POSITIVE_INFINITY) -
    76→    76→    (b.distanceKm ?? Number.POSITIVE_INFINITY)
    77→    77→  );
    78→    78→}

   217→   217→  /* 0) Restaurant data — two read-only sources:
   218→   218→   *    - nearby feed: coordinate-bearing places, distance-annotated server-side;
   219→   219→   *    - full list:   every discoverable place, INCLUDING rows whose
   220→   220→   *      coordinates are still NULL ("Location pending verification").
   221→   221→   *    Neither source ever writes to the database. */
   222→   222→  const feed = useRestaurantFeed(origin);
   223→   223→  const allFeed = useAllRestaurants();
   224→   224→
   225→   225→  /* 1) Haversine distance from the active origin to every coordinate-bearing

   227→   227→   *    (NO distanceKm) and tracked separately as "pending". */
   228→   228→  const withDistance = useMemo(() => withDistances(feed.places, origin), [feed.places, origin]);
   229→   229→  const allAnnotated = useMemo(
   230→   230→    () =>
   231→   231→      allFeed.places.map((p) =>
   232→   232→        p.latitude != null && p.longitude != null
   233→   233→          ? { ...p, distanceKm: haversineKm(origin, { lat: p.latitude, lng: p.longitude }) }
   234→   234→          : p
   235→   235→      ),
   236→   236→    [allFeed.places, origin]
   237→   237→  );
   238→   238→
   239→   239→  /* Merge by id — nearby results win (authoritative distance source). */
   240→   240→  const merged = useMemo(() => {
   241→   241→    const byId = new Map<string, PlaceMaybeWithDistance>();
   242→   242→    for (const p of allAnnotated) byId.set(p.id, p);
   243→   243→    for (const p of withDistance) byId.set(p.id, p);
   244→   244→    return [...byId.values()];
   245→   245→  }, [allAnnotated, withDistance]);
   246→   246→
   247→   247→  /* Places with a KNOWN distance — the only ones eligible for radius
   248→   248→   * filters and "nearest" sorting. */
   249→   249→  const withKnownDistance = useMemo(
   250→   250→    () => merged.filter((p): p is PlaceWithDistance => p.distanceKm != null),
   251→   251→    [merged]
   252→   252→  );
   253→   253→
   254→   254→  /* NULL-coordinate places — honest "Location pending verification" rows.
   255→   255→   * They carry no distance and never fake one. */
   256→   256→  const pending = useMemo(() => merged.filter((p) => p.distanceKm == null), [merged]);
   257→   257→
   258→   258→  /* 2) Distance filter applies ONLY to places with known distances. */
   259→   259→  const inRange = useMemo(
   260→   260→    () =>
   261→   261→      distanceFilterKm == null
   262→   262→        ? withKnownDistance
   263→   263→        : withKnownDistance.filter((p) => p.distanceKm <= distanceFilterKm),
   264→   264→    [withKnownDistance, distanceFilterKm]
   265→   265→  );
   266→   266→
   267→   267→  /* 3) Sort + section the filtered list. */
   268→   268→  const nearby = useMemo(
   269→   269→    () => (sortBy === "nearest" ? [...inRange].sort(byDistance) : [...inRange].sort(byRating)),
   270→   270→    [inRange, sortBy]
   271→   271→  );

   285→   285→      list = list.filter((p) => p.categoryIds.includes(activeCategory));

   289→   289→    }

   291→   291→  }, [inRange, pending, distanceFilterKm, activeCategory, searchQuery, sortBy]);

   293→   293→  /* Page-level loading / error across BOTH feeds (never a silent partial). */

   296→   296→    feed.status === "error" ? feed.error : allFeed.status === "error" ? allFeed.error : null;
   297→   297→  const retryFeed = () => {
   298→   298→    feed.reload();
   299→   299→    allFeed.reload();
   300→   300→  };

   302→   302→  const discoveryEmpty =
   303→   303→    nearby.length === 0 &&
   304→   304→    gems.length === 0 &&
   305→   305→    favorites.length === 0 &&
   306→   306→    popular.length === 0 &&
   307→   307→    budget.length === 0 &&
   308→   308→    // pending places count as content only when no radius filter hides them

   310→   310→
   311→   311→  /* The backend answered, but the database truly has zero restaurants. */
   312→   312→  const dbEmpty =
   313→   313→    feed.status === "ready" && allFeed.status === "ready" && merged.length === 0;
   314→   314→
   315→   315→  const demoBanner = feed.mode === "demo" || allFeed.mode === "demo";
   316→   316→  const demoNotice = feed.notice ?? allFeed.notice;
