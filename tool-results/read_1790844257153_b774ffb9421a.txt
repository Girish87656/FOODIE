    59→function greeting(): string {
    60→  const h = new Date().getHours();
    61→  if (h < 5) return "Late night cravings";
    62→  if (h < 12) return "Good morning";
    63→  if (h < 17) return "Good afternoon";
    64→  return "Good evening";
    65→}

    67→/**
    68→ * Distance sort that tolerates places WITHOUT a distance (NULL coordinates):
    69→ * they sink to the end of the list instead of faking a number. Only places
    70→ * with a real Haversine distance can ever rank first.
    71→ */
    72→function byKnownDistance(a: PlaceMaybeWithDistance, b: PlaceMaybeWithDistance): number {
    73→  return (
    74→    (a.distanceKm ?? Number.POSITIVE_INFINITY) -
    75→    (b.distanceKm ?? Number.POSITIVE_INFINITY)
    76→  );
    77→}

   189→/** Screen 5 — the discovery home feed, powered by the user's real location */
   190→export function HomeScreen() {
   191→  const go = useFoodieStore((s) => s.go);
   192→  const userName = useFoodieStore((s) => s.userName);
   193→  const eatContext = useFoodieStore((s) => s.eatContext);
   194→  const preferences = useFoodieStore((s) => s.preferences);
   195→  const searchQuery = useFoodieStore((s) => s.searchQuery);
   196→  const setSearchQuery = useFoodieStore((s) => s.setSearchQuery);
   197→  const activeCategory = useFoodieStore((s) => s.activeCategory);
   198→  const setActiveCategory = useFoodieStore((s) => s.setActiveCategory);
   199→  const openPlace = useFoodieStore((s) => s.openPlace);
   200→  const distanceFilterKm = useFoodieStore((s) => s.distanceFilterKm);
   201→  const setDistanceFilter = useFoodieStore((s) => s.setDistanceFilter);
   202→  const sortBy = useFoodieStore((s) => s.sortBy);
   203→  const setSortBy = useFoodieStore((s) => s.setSortBy);
   204→
   205→  // --- real location state (native Geolocation API → Zustand) ---
   206→  const headline = useLocationHeadline();
   207→  const locSource = useLocationStore((s) => s.source);
   208→  const locStatus = useLocationStore((s) => s.status);
   209→  const origin = useLocationStore((s) => s.origin);
   210→
   211→  const [expanded, setExpanded] = useState<string | null>(null);
   212→  const [locSheetOpen, setLocSheetOpen] = useState(false);
   213→
   214→  const searching = searchQuery.trim().length > 0 || activeCategory !== null;
   215→
   216→  /* 0) Restaurant data — two read-only sources:
   217→   *    - nearby feed: coordinate-bearing places, distance-annotated server-side;
   218→   *    - full list:   every discoverable place, INCLUDING rows whose
   219→   *      coordinates are still NULL ("Location pending verification").
   220→   *    Neither source ever writes to the database. */
   221→  const feed = useRestaurantFeed(origin);
   222→  const allFeed = useAllRestaurants();
   223→
   224→  /* 1) Haversine distance from the active origin to every coordinate-bearing
   225→   *    place. NULL-coordinate rows never enter this math — they are kept as-is
   226→   *    (NO distanceKm) and tracked separately as "pending". */
   227→  const withDistance = useMemo(() => withDistances(feed.places, origin), [feed.places, origin]);
   228→  const allAnnotated = useMemo(
   229→    () =>
   230→      allFeed.places.map((p) =>
   231→        p.latitude != null && p.longitude != null
   232→          ? { ...p, distanceKm: haversineKm(origin, { lat: p.latitude, lng: p.longitude }) }
   233→          : p
   234→      ),
   235→    [allFeed.places, origin]
   236→  );
   237→
   238→  /* Merge by id — nearby results win (authoritative distance source). */
   239→  const merged = useMemo(() => {
   240→    const byId = new Map<string, PlaceMaybeWithDistance>();
   241→    for (const p of allAnnotated) byId.set(p.id, p);
   242→    for (const p of withDistance) byId.set(p.id, p);
   243→    return [...byId.values()];
   244→  }, [allAnnotated, withDistance]);
   245→
   246→  /* Places with a KNOWN distance — the only ones eligible for radius
   247→   * filters and "nearest" sorting. */
   248→  const withKnownDistance = useMemo(
   249→    () => merged.filter((p): p is PlaceWithDistance => p.distanceKm != null),
   250→    [merged]
   251→  );
   252→
   253→  /* NULL-coordinate places — honest "Location pending verification" rows.
   254→   * They carry no distance and never fake one. */
   255→  const pending = useMemo(() => merged.filter((p) => p.distanceKm == null), [merged]);
   256→
   257→  /* 2) Distance filter applies ONLY to places with known distances. */
   258→  const inRange = useMemo(
   259→    () =>
   260→      distanceFilterKm == null
   261→        ? withKnownDistance
   262→        : withKnownDistance.filter((p) => p.distanceKm <= distanceFilterKm),
   263→    [withKnownDistance, distanceFilterKm]
   264→  );
   265→
   266→  /* 3) Sort + section the filtered list. */
   267→  const nearby = useMemo(
   268→    () => (sortBy === "nearest" ? [...inRange].sort(byDistance) : [...inRange].sort(byRating)),
   269→    [inRange, sortBy]
   270→  );
   271→  const gems = useMemo(() => hiddenGems(inRange), [inRange]);
   272→  const favorites = useMemo(() => localFavorites(inRange), [inRange]);
   273→  const popular = useMemo(() => popularPlaces(inRange), [inRange]);
   274→  const budget = useMemo(() => budgetEats(inRange), [inRange]);
   275→
   276→  /* 4) Search / category results — pending places join the pool whenever no
   277→   *    radius filter is active (an unknown distance can't honestly satisfy
   278→   *    "within X km"). With "Nearest First" they sort to the very end. */
   279→  const results = useMemo(() => {
   280→    const pool: PlaceMaybeWithDistance[] =
   281→      distanceFilterKm == null ? [...inRange, ...pending] : [...inRange];
   282→    let list = pool;
   283→    if (activeCategory) {
   284→      list = list.filter((p) => p.categoryIds.includes(activeCategory));
   285→    }
   286→    if (searchQuery.trim()) {
   287→      list = list.filter((p) => matchesQuery(p, searchQuery));
   288→    }
   289→    return sortBy === "nearest" ? [...list].sort(byKnownDistance) : [...list].sort(byRating);
   290→  }, [inRange, pending, distanceFilterKm, activeCategory, searchQuery, sortBy]);
   291→
   292→  /* Page-level loading / error across BOTH feeds (never a silent partial). */
   293→  const feedLoading = feed.status === "loading" || allFeed.status === "loading";
   294→  const feedError =
   295→    feed.status === "error" ? feed.error : allFeed.status === "error" ? allFeed.error : null;
   296→  const retryFeed = () => {
   297→    feed.reload();
   298→    allFeed.reload();
   299→  };
   300→
   301→  const discoveryEmpty =
   302→    nearby.length === 0 &&
   303→    gems.length === 0 &&
   304→    favorites.length === 0 &&
   305→    popular.length === 0 &&
   306→    budget.length === 0 &&
   307→    // pending places count as content only when no radius filter hides them
   308→    (distanceFilterKm != null || pending.length === 0);
   309→
   310→  /* The backend answered, but the database truly has zero restaurants. */
   311→  const dbEmpty =
   312→    feed.status === "ready" && allFeed.status === "ready" && merged.length === 0;
   313→
   314→  const demoBanner = feed.mode === "demo" || allFeed.mode === "demo";
   315→  const demoNotice = feed.notice ?? allFeed.notice;
   316→
   317→  const preferredCats = PLACE_CATEGORIES.filter((c) => preferences.includes(c.id));
   318→
   319→  const nearbySubtitle =
   320→    locSource === "browser"
   321→      ? "Closest to your live location"
   322→      : locSource === "manual"
   323→        ? "Distances from your selected city"
   324→        : locSource === "test"
   325→          ? `Distances from ${GITAM_TEST_LOCATION.label}`
   326→          : `Around ${PROTOTYPE_CITY.area} · prototype origin`;
   327→
   328→  function toggleExpand(key: string) {
   329→    setExpanded((cur) => (cur === key ? null : key));
   330→  }
   331→
   332→  return (
   333→    <div className="relative flex h-full flex-col bg-background">
   334→      {/* ---------------- header ---------------- */}
   335→      <header className="z-10 shrink-0 border-b border-border/50 bg-background/95 px-5 pb-3 pt-5 backdrop-blur supports-[backdrop-filter]:bg-background/80">
   336→        <div className="flex items-center justify-between gap-3">
   337→          <button
   338→            type="button"
   339→            onClick={() => setLocSheetOpen(true)}
   340→            aria-haspopup="dialog"
   341→            className="group flex min-w-0 items-start gap-1.5 text-left"
   342→          >
   343→            <span className="relative mt-0.5 shrink-0">
   344→              <MapPinned
   345→                className={cn(
   346→                  "size-4",
   347→                  headline.tone === "live" ? "text-emerald-600" : "text-primary"
   348→                )}
   349→                aria-hidden="true"
   350→              />
   351→              {headline.tone === "live" && (
   352→                <span
   353→                  className="absolute -right-0.5 -top-0.5 size-1.5 animate-pulse rounded-full bg-emerald-500"
   354→                  aria-hidden="true"
   355→                />
   356→              )}
   357→            </span>
   358→            <span className="min-w-0">
   359→              <span
   360→                className={cn(
   361→                  "block text-[10.5px] font-bold uppercase tracking-wider",
   362→                  headline.tone === "live"
